; lower address byte for lower bits
; higher address byte for higher bits
; store all variables on the 0800 page


; initialize integer (0800, 0801)
  lda $00
  sta @0800
  sta @0801

; initialize integer (0802, 0803)
  lda $00
  sta @0802
  sta @0803

  ldx $04
  xlx  
  add &0800
  ldx $05
  xlx  
  sta &0800
